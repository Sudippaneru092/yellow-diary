-- 1. Users Table
CREATE TABLE Users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(255),
    email VARCHAR(150) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    user_type ENUM('Admin', 'General', 'Guest') DEFAULT 'General',
    total_time_spent INT DEFAULT 0,
    last_login_time DATETIME,
    is_active BOOLEAN DEFAULT TRUE,
    avatar VARCHAR(255) DEFAULT NULL
);

-- 2. Businesses Table
CREATE TABLE Businesses (
    BusinessID INT AUTO_INCREMENT PRIMARY KEY,
    OwnerID INT NOT NULL,
    BusinessName VARCHAR(150) NOT NULL,
    Description TEXT,
    Address VARCHAR(255),
    City VARCHAR(100),
    District VARCHAR(100),
    ContactInfo VARCHAR(150),
    Approved BOOLEAN DEFAULT FALSE,
    CreatedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (OwnerID) REFERENCES Users(id) ON DELETE CASCADE
);

-- 3. Categories Table
CREATE TABLE Categories (
    CategoryID INT AUTO_INCREMENT PRIMARY KEY,
    CategoryName VARCHAR(100) UNIQUE NOT NULL
);

-- 4. BusinessCategories Junction Table
CREATE TABLE BusinessCategories (
    BusinessID INT NOT NULL,
    CategoryID INT NOT NULL,
    PRIMARY KEY (BusinessID, CategoryID),
    FOREIGN KEY (BusinessID) REFERENCES Businesses(BusinessID) ON DELETE CASCADE,
    FOREIGN KEY (CategoryID) REFERENCES Categories(CategoryID) ON DELETE CASCADE
);

-- 5. Posts Table (for jobs, events, announcements)
CREATE TABLE Posts (
    PostID INT AUTO_INCREMENT PRIMARY KEY,
    BusinessID INT NOT NULL,
    Title VARCHAR(150) NOT NULL,
    Content TEXT NOT NULL,
    PostType ENUM('Job', 'Event', 'Announcement') NOT NULL,
    Approved BOOLEAN DEFAULT FALSE,
    CreatedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (BusinessID) REFERENCES Businesses(BusinessID) ON DELETE CASCADE
);

-- 6. Admins Table (optional — only if you need separate logic from Users)
CREATE TABLE Admins (
    AdminID INT PRIMARY KEY,
    Privileges TEXT,
    FOREIGN KEY (AdminID) REFERENCES Users(id) ON DELETE CASCADE
);

-- 7. SearchLogs Table
CREATE TABLE SearchLogs (
    SearchID INT AUTO_INCREMENT PRIMARY KEY,
    UserID INT,
    SearchQuery VARCHAR(255),
    SearchDateTime TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (UserID) REFERENCES Users(id) ON DELETE SET NULL
);
